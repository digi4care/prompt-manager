<task type="auto">
  <name>Task 2: Create execution-panel.svelte with state machine</name>
  <files>src/lib/components/prompts/execution-panel.svelte</files>
  <action>
Create the main execution panel with state machine pattern:

1. **Props interface**:

   ```typescript
   interface Props {
   	promptId: number;
   	content: string; // The prompt content to execute
   	functionType?: 'executor' | 'judge' | 'improve' | 'council';
   	class?: string;
   }
   ```

2. **State machine**:

   ```typescript
   type ExecutionState = 'idle' | 'loading' | 'success' | 'error';

   let state = $state<ExecutionState>('idle');
   let result = $state<ExecutionResult | null>(null);
   let error = $state<{ message: string; code?: string; recovery?: string } | null>(null);
   let overrides = $state<RunOverrides>({});
   ```

3. **Derived states**:

   ```typescript
   let isExecuting = $derived(state === 'loading');
   let canExecute = $derived(state !== 'loading' && content.trim().length > 0);
   let hasResult = $derived(state === 'success' && result !== null);
   let hasError = $derived(state === 'error' && error !== null);
   ```

4. **Execute function**:

   ```typescript
   async function handleExecute() {
   	if (!canExecute) return;

   	state = 'loading';
   	error = null;
   	result = null;

   	try {
   		const response = await fetch(`/api/prompts/${promptId}/execute`, {
   			method: 'POST',
   			headers: { 'Content-Type': 'application/json' },
   			body: JSON.stringify({
   				content,
   				overrides: Object.keys(overrides).length > 0 ? overrides : undefined
   			})
   		});

   		if (!response.ok) {
   			const data = await response.json();
   			throw {
   				message: data.message || 'Execution failed',
   				code: data.code,
   				recovery: data.recovery
   			};
   		}

   		result = await response.json();
   		state = 'success';
   	} catch (e) {
   		error =
   			e instanceof Error
   				? { message: e.message }
   				: (e as { message: string; code?: string; recovery?: string });
   		state = 'error';
   	}
   }
   ```

5. **Retry and reset functions**:

   ```typescript
   function handleRetry() {
   	handleExecute();
   }

   function handleReset() {
   	state = 'idle';
   	result = null;
   	error = null;
   }
   ```

6. **Layout structure**:

   ```svelte
   <div class="space-y-4 {className}">
   	<!-- Override controls (collapsible) -->
   	<ExecutionOverrides
   		{overrides}
   		onchange={(o) => (overrides = o)}
   		defaults={{ modelId: 'default', temperature: 0.7, maxTokens: 4096 }}
   		disabled={isExecuting}
   	/>

   	<!-- Execute button -->
   	<Button onclick={handleExecute} disabled={!canExecute} class="w-full">
   		{#if isExecuting}
   			<Loader2 class="mr-2 h-4 w-4 animate-spin" />
   			Executing...
   		{:else}
   			<Play class="mr-2 h-4 w-4" />
   			Execute
   		{/if}
   	</Button>

   	<!-- Error display -->
   	{#if hasError}
   		<div class="rounded-lg border border-red-200 bg-red-50 p-4">
   			<div class="flex items-start gap-3">
   				<AlertCircle class="mt-0.5 h-5 w-5 text-red-500" />
   				<div class="flex-1">
   					<p class="font-medium text-red-800">{error.message}</p>
   					{#if error.recovery}
   						<p class="mt-1 text-sm text-red-600">{error.recovery}</p>
   					{/if}
   				</div>
   			</div>
   			<div class="mt-3 flex gap-2">
   				<Button variant="outline" size="sm" onclick={handleRetry}>Retry</Button>
   				<Button variant="ghost" size="sm" onclick={handleReset}>Dismiss</Button>
   			</div>
   		</div>
   	{/if}

   	<!-- Success result -->
   	{#if hasResult}
   		<ExecutionResult
   			{result}
   			oncopy={() => {
   				/* optional toast */
   			}}
   		/>
   		<Button variant="outline" size="sm" onclick={handleReset}>Clear Result</Button>
   	{/if}
   </div>
   ```

7. **Imports**:
   - Button from `$lib/components/ui/button`
   - ExecutionOverrides from `./execution-overrides.svelte`
   - ExecutionResult from `./execution-result.svelte`
   - Icons: Play, Loader2, AlertCircle from `@lucide/svelte`
   - RunOverrides type from settings-cascade.service
