2. State machine (use Svelte 5 runes):

```typescript
type CouncilUIState = 'idle' | 'running' | 'complete' | 'error';
type ActiveStep = 'producer' | 'reviewer' | 'fixer' | null;

let uiState = $state<CouncilUIState>('idle');
let activeStep = $state<ActiveStep>(null);
let currentRound = $state(1);
let stepResults = $state<StepResult[]>([]);
let finalOutput = $state<string | null>(null);
let errorMessage = $state<string | null>(null);
let streamContent = $state(''); // Current streaming content
```

3. Derived progress calculation:

```typescript
let progress = $derived(
	uiState === 'complete'
		? 100
		: uiState === 'idle'
			? 0
			: (((currentRound - 1) * 3 + stepIndex(activeStep)) / 9) * 100
);

function stepIndex(step: ActiveStep): number {
	return step === 'producer' ? 1 : step === 'reviewer' ? 2 : step === 'fixer' ? 3 : 0;
}
```

4. SSE connection using sveltekit-sse source():

```typescript
function startCouncil() {
	uiState = 'running';
	stepResults = [];
	currentRound = 1;
	streamContent = '';

	const connection = source(`/api/council/correct`, {
		options: {
			method: 'POST',
			body: JSON.stringify({ promptId, content })
		},
		open({ status }) {
			/* reset reconnect attempts */
		},
		close({ connect, isLocal }) {
			/* handle reconnection */
		},
		error({ error }) {
			/* handle error */
		}
	});

	// Use $effect to subscribe to each event type
}
```

5. Event subscriptions with $effect:

```typescript
$effect(() => {
	if (!connection) return;

	// step_start events
	const starts = connection.select('step_start');
	const unsub1 = starts.subscribe((data) => {
		const event = JSON.parse(data);
		activeStep = event.step;
		currentRound = event.round;
		streamContent = '';
	});

	// step_delta events
	const deltas = connection.select('step_delta');
	const unsub2 = deltas.subscribe((data) => {
		const event = JSON.parse(data);
		streamContent = event.data?.delta || '';
	});

	// step_complete events
	const completes = connection.select('step_complete');
	const unsub3 = completes.subscribe((data) => {
		const event = JSON.parse(data);
		stepResults = [
			...stepResults,
			{
				step: event.step,
				round: event.round,
				output: event.data?.output,
				issues: event.data?.issues,
				model: event.data?.model,
				usage: event.data?.usage
			}
		];
	});

	// round_complete events
	const rounds = connection.select('round_complete');
	const unsub4 = rounds.subscribe((data) => {
		currentRound++;
	});

	// council_complete events
	const done = connection.select('council_complete');
	const unsub5 = done.subscribe((data) => {
		const event = JSON.parse(data);
		finalOutput = event.data?.finalOutput;
		uiState = 'complete';
		activeStep = null;
		connection.close();
	});

	// error events
	const errors = connection.select('error');
	const unsub6 = errors.subscribe((data) => {
		const event = JSON.parse(data);
		errorMessage = event.message;
		uiState = 'error';
	});

	return () => {
		unsub1();
		unsub2();
		unsub3();
		unsub4();
		unsub5();
		unsub6();
	};
});
```

6. UI Layout:

```svelte
<div class="council-panel space-y-4">
	<!-- Progress bar -->
	<div class="progress-bar-container">
		<div class="progress-bar" style="width: {progress}%"></div>
		<span class="progress-text">Round {currentRound}/3 - {activeStep || 'Starting'}</span>
	</div>

	<!-- Step results (completed steps) -->
	{#each stepResults as result, i}
		<div
			class="step-result"
			class:producer={result.step === 'producer'}
			class:reviewer={result.step === 'reviewer'}
			class:fixer={result.step === 'fixer'}
		>
			<div class="step-header">
				<span class="badge">Round {result.round}</span>
				<span class="step-name">{result.step}</span>
				{#if result.step === 'reviewer' && result.issues}
					<span class="issues-count">{result.issues.length} issues</span>
				{/if}
			</div>
			<div class="step-output prose prose-sm">{result.output}</div>
		</div>
	{/each}

	<!-- Active step streaming -->
	{#if uiState === 'running' && activeStep}
		<div class="step-streaming">
			<div class="step-header">
				<span class="badge">Round {currentRound}</span>
				<span class="step-name">{activeStep}</span>
				<Loader2 class="animate-spin" />
			</div>
			<div class="stream-content prose prose-sm">
				{streamContent || 'Starting...'}
				<span class="cursor animate-pulse">▊</span>
			</div>
		</div>
	{/if}

	<!-- Final output -->
	{#if uiState === 'complete' && finalOutput}
		<div class="final-output">
			<h3>Final Output</h3>
			<div class="prose">{finalOutput}</div>
		</div>
	{/if}

	<!-- Error state -->
	{#if uiState === 'error'}
		<div class="error-panel">
			<AlertCircle />
			<p>{errorMessage}</p>
			<Button onclick={startCouncil}>Retry</Button>
		</div>
	{/if}

	<!-- Controls -->
	{#if uiState === 'idle'}
		<Button onclick={startCouncil} disabled={!content.trim()}>
			<Play /> Run Council Correct
		</Button>
	{/if}

	{#if uiState === 'complete' || uiState === 'error'}
		<Button
			variant="outline"
			onclick={() => {
				uiState = 'idle';
				stepResults = [];
				finalOutput = null;
			}}
		>
			Reset
		</Button>
	{/if}
</div>
```
