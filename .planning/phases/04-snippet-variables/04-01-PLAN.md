<behavior>
Extract variable names from template content using regex `/{{\s*([A-Za-z_][A-Za-z0-9_]*)\s*}}/g`.
Parse variable definitions from frontmatter YAML with Zod validation.

Cases:

- "Hello {{NAME}}" -> extracts ["NAME"]
- "No variables here" -> extracts []
- "{{FIRST}} and {{SECOND}}" -> extracts ["FIRST", "SECOND"]
- "{{  SPACED  }}" -> extracts ["SPACED"] (whitespace tolerant)
- "{{1INVALID}}" -> not matched (must start with letter or underscore)
- Duplicate {{NAME}} {{NAME}} -> extracts ["NAME"] (deduplicated)
  </behavior>

   - description: optional string
   - default: optional string
   - required: boolean (default true)
3. Create ExtractedVariable interface with name, startIndex, endIndex
4. Implement extractVariables(content: string): ExtractedVariable[]
   - Use regex `/{{\s*([A-Za-z_][A-Za-z0-9_]*)\s*}}/g`

   - Reset regex.lastIndex = 0 before returning

1. Zod schema for SnippetVariable:

```typescript
import { z } from 'zod';

export const VARIABLE_REGEX = /{{\s*([A-Za-z_][A-Za-z0-9_]*)\s*}}/g;

export const SnippetVariableSchema = z.object({
	name: z.string().regex(/^[A-Za-z_][A-Za-z0-9_]*$/, 'Invalid variable name'),
	description: z.string().optional(),
	default: z.string().optional(),
	required: z.boolean().default(true)
});

export type SnippetVariable = z.infer<typeof SnippetVariableSchema>;

export interface ExtractedVariable {
	name: string;
	startIndex: number;
	endIndex: number;
}

export function extractVariables(content: string): ExtractedVariable[] {
	const variables: ExtractedVariable[] = [];
	const seen = new Set<string>();
	const regex = new RegExp(VARIABLE_REGEX.source, 'g');
	let match: RegExpExecArray | null;

	while ((match = regex.exec(content)) !== null) {
		if (!seen.has(match[1])) {
			seen.add(match[1]);
			variables.push({
				name: match[1],
				startIndex: match.index,
				endIndex: match.index + match[0].length
			});
		}
	}

	return variables;
}
```

export type SnippetParseResult = FrontmatterParseResult & {
variables: SnippetVariable[];
};

// Add new function
export function parseSnippetFrontmatter(
frontmatterYaml: string | null | undefined
): SnippetParseResult {
const base = parseFrontmatterYaml(frontmatterYaml);
const variables: SnippetVariable[] = [];

if (base.extras.variables && Array.isArray(base.extras.variables)) {
for (const v of base.extras.variables) {
const result = SnippetVariableSchema.safeParse(v);
if (result.success) {
variables.push(result.data);
}
}
}

return { ...base, variables };
}
