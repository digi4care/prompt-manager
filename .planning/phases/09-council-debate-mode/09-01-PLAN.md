**Types to define:**

```typescript
export type DebateArchetype = 'proponent' | 'skeptic' | 'pragmatist';
export type DebateState =
	| 'idle'
	| 'debating_round_1'
	| 'debating_round_2'
	| 'debating_round_3'
	| 'synthesizing'
	| 'complete'
	| 'error';

export interface DebateAgentConfig {
	id: number;
	archetype: DebateArchetype;
	name: string;
	systemPrompt: string;
	modelId: string;
	providerId: string;
}

export interface DebateAgentResult {
	archetype: DebateArchetype;
	agentName: string;
	output: string;
	model: { providerId: string; modelId: string };
	usage: { promptTokens: number; completionTokens: number };
}

export interface DebateRoundResult {
	round: number;
	agentResults: DebateAgentResult[];
}

export interface DebateSynthesis {
	summary: string;
	keyArgumentsFor: string[];
	keyArgumentsAgainst: string[];
	pointsOfAgreement: string[];
	finalRecommendation: string;
	consensusLevel: 'Strong consensus' | 'Moderate consensus' | 'Mixed views';
}

export interface DebateRun {
	id: number;
	promptId: number;
	topicContent: string;
	currentRound: number;
	state: DebateState;
	rounds: DebateRoundResult[];
	synthesis?: DebateSynthesis;
	createdAt: Date;
	updatedAt: Date;
}
```

**Endpoint: POST /api/council/debate**

**Request validation (Zod):**

```typescript
const DebateRequestSchema = z.object({
	promptId: z.number().int().positive(),
	topic: z.string().min(1).max(50000),
	agentOverrides: z
		.record(
			z.object({
				promptId: z.number().int().positive(),
				versionId: z.number().int().positive().optional()
			})
		)
		.optional()
});
```

**Implementation:**

1. Parse and validate request body
2. Call `authenticateRequest(event)` for JWT auth
3. Convert agentOverrides from record to Map for service
4. Call `executeDebate()` generator
5. Use `produce()` from sveltekit-sse to stream events
6. Serialize each DebateEvent as JSON via `JSON.stringify(event)`
7. Include 15-second heartbeat via onOpen callback
8. Handle errors gracefully with error event emission

**Event stream format:**

```
event: message
data: {"type":"debate_start","data":{"topic":"...","maxRounds":3}}

event: message
data: {"type":"round_start","round":1,"data":{"round":1}}

event: message
data: {"type":"agent_delta","round":1,"archetype":"proponent","data":{"delta":"..."}}

event: message
data: {"type":"debate_complete","data":{"rounds":[...],"synthesis":{...}}}
```

**Error handling:**

- Validation errors: return 400 with error details
- Auth errors: return 401
- Execution errors: emit error event then close stream
